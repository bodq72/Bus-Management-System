        <input type="text" id="user" placeholder="Username (admin)">
        <input type="password" id="pass" placeholder="Password (admin123)">
        <button onclick="handleLogin()">Login</button>
        <button style="background:#7f8c8d;" onclick="handleSignup()">Sign Up</button>
    </div>

    <!-- APP CONTENT -->
    <div id="app-view" class="hidden">
        <p id="welcome" style="text-align:right; font-size:12px;"></p>
        
        <div id="admin-tools" class="hidden">
            <h4 style="margin-bottom:5px;">Admin: Add Bus</h4>
            <input type="number" id="bus-id" placeholder="Bus Number">
            <input type="text" id="bus-route" placeholder="Routes (e.g. Cairo, Giza, Alex)">
            <button style="background:#27ae60;" onclick="addBus()">Save Bus</button>
            <hr>
        </div>

        <h4>Search Route</h4>
        <input type="text" id="from-loc" placeholder="From...">
        <input type="text" id="to-loc" placeholder="To...">
        <button onclick="searchBuses()">Find Bus</button>
        <button style="background:#9b59b6;" onclick="renderList()">Show All</button>
        
        <div id="results" style="margin-top:20px;"></div>
        <button style="background:#e74c3c; margin-top:30px;" onclick="location.reload()">Logout</button>
    </div>

<script>
    // Data Storage
    let users = JSON.parse(localStorage.getItem('bus_users')) || [{u:'admin', p:'admin123', role:'admin'}];
    let buses = JSON.parse(localStorage.getItem('bus_list')) || [];
    let activeUser = null;

    function handleLogin() {
        const u = document.getElementById('user').value;
        const p = document.getElementById('pass').value;
        const found = users.find(x => x.u === u && x.p === p);

        if(found) {
            activeUser = found;
            document.getElementById('login-view').classList.add('hidden');
            document.getElementById('app-view').classList.remove('hidden');
            document.getElementById('welcome').innerText = "User: " + u;
            if(found.role === 'admin') document.getElementById('admin-tools').classList.remove('hidden');
        } else {
            document.getElementById('error-msg').innerText = "Invalid Login!";
        }
    }

    function handleSignup() {
        const u = document.getElementById('user').value;
        const p = document.getElementById('pass').value;
        if(!u || !p) return alert("Enter details");
        if(users.find(x => x.u === u)) return alert("User exists");

        users.push({u: u, p: p, role: 'user'});
        localStorage.setItem('bus_users', JSON.stringify(users));
        alert("Signed up! Now click Login.");
    }

    function addBus() {
        const id = document.getElementById('bus-id').value;
        const routeStr = document.getElementById('bus-route').value;
        if(!id || !routeStr) return;

        const routeArr = routeStr.split(',').map(s => s.trim().toUpperCase());
        buses.push({id: id, stops: routeArr});
        localStorage.setItem('bus_list', JSON.stringify(buses));
        alert("Bus " + id + " Added");
        document.getElementById('bus-id').value = '';
        document.getElementById('bus-route').value = '';
    }

    function searchBuses() {
        const f = document.getElementById('from-loc').value.toUpperCase();
        const t = document.getElementById('to-loc').value.toUpperCase();
        const out = document.getElementById('results');
        
        const match = buses.filter(b => b.stops.includes(f) && b.stops.includes(t));
        out.innerHTML = "<b>Results:</b><br>";
        if(match.length === 0) out.innerHTML += "None found.";
        match.forEach(m => {
            out.innerHTML += `<div class="bus-item">🚌 Bus <b>${m.id}</b> serves this route!</div>`;
        });
    }

    function renderList() {
        const out = document.getElementById('results');
        out.innerHTML = "<b>All Buses:</b>";
        buses.forEach(b => {
            out.innerHTML += `<div class="bus-item"><b>No. ${b.id}</b>: ${b.stops.join(' → ')}</div>`;
        });
    }
</script>
